## Example automated tests (pytest + requests)

```python
import io
import hashlib
import pytest
import requests

BASE = "http://localhost:8000"
UPLOAD = f"{BASE}/api/upload"
MAX_SIZE = 5 * 1024 * 1024  # 5 MB
HEADERS = {"Authorization": "Bearer <test-token>"}


def upload(name, content, mime="application/octet-stream", headers=HEADERS):
    return requests.post(
        UPLOAD,
        files={"file": (name, io.BytesIO(content), mime)},
        headers=headers,
    )


def test_valid_png_upload_succeeds():
    png = b"\x89PNG\r\n\x1a\n" + b"\x00" * 100
    r = upload("photo.png", png, "image/png")
    assert r.status_code in (200, 201)
    assert "id" in r.json()


def test_uploaded_file_content_is_intact():
    data = b"hello world" * 1000
    r = upload("note.txt", data, "text/plain")
    file_id = r.json()["id"]
    downloaded = requests.get(f"{BASE}/api/files/{file_id}", headers=HEADERS).content
    assert hashlib.sha256(downloaded).hexdigest() == hashlib.sha256(data).hexdigest()


def test_empty_file_rejected():
    assert upload("empty.txt", b"", "text/plain").status_code == 400


def test_file_at_size_limit_accepted():
    assert upload("max.txt", b"a" * MAX_SIZE, "text/plain").status_code in (200, 201)


def test_file_over_size_limit_rejected():
    assert upload("big.txt", b"a" * (MAX_SIZE + 1), "text/plain").status_code == 413


@pytest.mark.parametrize("name", ["virus.exe", "shell.sh", "page.php", "a.jpg.php"])
def test_disallowed_extensions_rejected(name):
    assert upload(name, b"data").status_code in (400, 415)


def test_mime_spoofing_rejected():
    exe_header = b"MZ" + b"\x00" * 100  # Windows executable magic bytes
    assert upload("fake.jpg", exe_header, "image/jpeg").status_code in (400, 415)


def test_path_traversal_filename_is_sanitized():
    r = upload("../../etc/passwd", b"x", "text/plain")
    if r.status_code in (200, 201):
        assert ".." not in r.json().get("filename", "")


def test_missing_file_field_returns_400():
    assert requests.post(UPLOAD, headers=HEADERS).status_code == 400


def test_unauthenticated_upload_rejected():
    assert upload("a.txt", b"x", "text/plain", headers={}).status_code in (401, 403)
```
